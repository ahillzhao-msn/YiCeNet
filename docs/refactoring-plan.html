4. interfaces.py — 从 1 ABC 扩展到 6 个核心接口

from abc import ABC, abstractmethod
from pathlib import Path
import torch
import numpy as np

# ────────────── ITokenizer ──────────────
class ITokenizer(ABC):
    """Qwen 之外的 tokenizer 也可实现此接口。"""

    @abstractmethod
    def encode(self, text: str, max_len: int = 128
              ) -> tuple[torch.Tensor, torch.Tensor]:
        ...  # → (input_ids: (1,T), mask: (1,T))

    @abstractmethod
    def get_vocab_size(self) -> int:
        ...  # → 8000 (or whatever)

    @abstractmethod
    def download(self, hf_token: str = "") -> bool:
        ...

# ────────────── IEncoder ──────────────
class IEncoder(ABC):
    """可替换的编码器。默认 TinyEncoder。"""

    @abstractmethod
    def encode_context(
        self, input_ids: torch.Tensor, mask: torch.Tensor,
        env_vec: torch.Tensor | None = None,
    ) -> torch.Tensor:
        ...  # → h: (B, 256)

# ────────────── IRouter ──────────────
class IRouter(ABC):
    """可替换的路由策略。默认 GumbelRouter。"""

    @abstractmethod
    def divine(
        self, h: torch.Tensor, tau: float = 1.0, hard: bool = False,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        ... # → (hex_idx: (B,), probs: (B, 64))

# ────────────── IValueNetwork ──────────────
class IValueNetwork(ABC):
    """可替换的值网络。默认 3-layer MLP。"""

    @abstractmethod
    def score(
        self, candidate_embeds: torch.Tensor
    ) -> torch.Tensor:
        ...  # → values: (B, 8, 1)

# ────────────── IMemoryBank ──────────────
class IMemoryBank(ABC):
    """可注入的短期记忆。不再全局单例。"""

    @abstractmethod
    def init_session(self, session_id: str): ...

    @abstractmethod
    def store_turn(
        self, session_id: str, turn_id: int,
        encoder_output: np.ndarray, hexagram_id: int,
        summary: str = "",
    ): ...

    @abstractmethod
    def get_session_keys(
        self, session_id: str
    ) -> tuple[np.ndarray, list[dict]]:
        ... # → (keys: (N,384), meta: list[dict])

# ────────────── IEngine ──────────────
class IEngine(ABC):
    """核心推理引擎接口（不是 God Class，是门面）。"""

    @abstractmethod
    def predict(
        self, task_brief: str,
        temperature: float = 0.1,
        deterministic: bool = False,
        environment: dict | None = None,
    ) -> PredictionResult:
        ...  # 纯推理，不含上下文处方

    @abstractmethod
    def prescribe(
        self, task_text: str, session_id: str,
        turn_id: int = 0,
        turn_summary: str = "",
    ) -> Prescription:
        ...  # 上下文处方 = predict + memory + attention

    @abstractmethod
    def analyze(
        self, task_brief: str,
    ) -> EnvAnalysis:
        ...  # 纯环境分析 = encode + probe，不含路由

    @abstractmethod
    def switch_model(self, checkpoint: str) -> bool:
        ...

5. engine.py — predict() 分解为三步

# ── 不再在一个方法里做 7 件事 ──

class YiCeNetEngine:
    """构造时注入所有依赖，而非自 new。"""

    def __init__(
        self,
        model: YiCeNet,
        tokenizer: ITokenizer,
        memory_bank: IMemoryBank,           # ← 注入，不调 get_memory_bank()
        encoder: IEncoder | None = None,  # ← 可替换
        router: IRouter | None = None,      # ← 可替换
        value_net: IValueNetwork | None = None,
    ):
        self._model = model
        self._tokenizer = tokenizer
        self._memory = memory_bank
        self._encoder = encoder or model.encoder
        self._router = router or model.router
        self._value_net = value_net or model.value_net

    def predict(
        self, task_brief: str,
        temperature: float = 0.1,
        deterministic: bool = False,
        environment: dict | None = None,
    ) -> PredictionResult:
        # ① 编码
        input_ids, mask = self._tokenizer.encode(task_brief)
        env_vec = build_env_vec(environment)
        h = self._encoder.encode_context(input_ids, mask, env_vec)

        # ② 路由
        hex_idx, probs = self._router.divine(
            h, tau=temperature, hard=deterministic,
        )

        # ③ 候选评估
        cand_idxs = generate_candidates(hex_idx.item())
        cand_embeds = self._model.hexagram_embed(cand_idxs)
        cand_values = self._value_net.score(cand_embeds)
        best_idx = cand_values.argmax(dim=-1)
        best_hex_id = cand_idxs.gather(1, best_idx)

        # ④ 动作
        action_ids = self._model.decode_action(best_hex_id)

        # ⑤ 探针
        probes = extract_probes_tensor(
            h, ..., hex_idx, self._prev_hx, ...,
        )
        self._prev_hx = h

        # ⑥ 信心评估
        confidence, status, hint = compute_env_confidence(
            probes.tolist(), cand_values[0].tolist(),
        )

        return PredictionResult(
            hexagram_id=hex_idx.item(),
            hexagram_name=HEXAGRAM_NAMES[hex_idx.item()],
            selected_hexagram_id=best_hex_id.item(),
            selected_hexagram_name=HEXAGRAM_NAMES[best_hex_id.item()],
            candidates=[...],
            action_id=action_ids.item(),
            probes=probes.tolist(),
            env_confidence=confidence,
            context_status=status,
            context_hint=hint,
            ...
        )

    def prescribe(
        self, task_text: str, session_id: str,
        turn_id: int = 0, turn_summary: str = "",
    ) -> Prescription:
        # predict + memory + attention — 独立方法，不混在 predict() 里
        result = self.predict(task_text)
        h, mask, env_vec = ...  # 复用编码结果
        self._memory.store_turn(
            session_id, turn_id, h.numpy(), result["hexagram_id"], turn_summary,
        )
        keys, meta = self._memory.get_session_keys(session_id)
        weights = CrossAttention.compute(h.numpy().squeeze(), keys)
        prescription = ContextPrescription.generate(
            weights, meta, result["probes"],
        )
        return prescription

    def analyze(
        self, task_brief: str,
    ) -> EnvAnalysis:
        # 纯编码 + 探针，不走路由。用于快速环境感知（~3ms）。
        input_ids, mask = self._tokenizer.encode(task_brief)
        h = self._encoder.encode_context(input_ids, mask)
        probes = extract_probes_tensor(h, ...)
        return EnvAnalysis(h=h, probes=probes, ...)

# ── 取代 hermes_tool._get_engine() + mcp_server._get_engine() ──

class EngineProvider:
    """统一 engine 生命周期管理。
    hermes_tool.py、mcp_server.py、测试代码全部走此提供者。
    """

    _engine: IEngine | None = None
    _active_version: str = ""

    @classmethod
    def get_engine(cls) -> IEngine:
        """懒加载 + 注册表发现 + 自动热切换。"""
        if cls._engine is None:
            cls._engine = cls._build_engine()
        return cls._engine

    @classmethod
    def check_switch(cls) -> bool:
        """注册表检查：ready 是否优于 active + 3% → 热切换。"""
        reg = load_registry()
        if promote_ready(reg):
            cls._engine = cls._build_engine(
                checkpoint=reg["ready"]["path"],
            )
            return True
        return False

    @classmethod
    def _build_engine(cls, checkpoint: str = "") -> IEngine:
        """一次封装：registry → model + tokenizer + memory → YiCeNetEngine"""
        ckpt_path, ver = resolve_checkpoint(checkpoint)
        model = YiCeNet.from_pretrained(ckpt_path)
        tokenizer = QwenTokenizer()      # 实现 ITokenizer
        memory = MemoryBank()             # 实现 IMemoryBank
        return YiCeNetEngine(
            model=model, tokenizer=tokenizer, memory_bank=memory,
        )

# install/base.py
class PlatformInstaller(ABC):
    """每个平台一个实现。"""

    @abstractmethod
    def detect() -> bool:        # 检测是否安装了本平台 (Hermes/Claude/None)
    @abstractmethod
    def install_package(): ...    # pip install 到该平台 venv
    @abstractmethod
    def register_hooks(): ...    # 注册 lifecycle hooks
    @abstractmethod
    def unregister(): ...        # 清理痕迹

# providers/__init__.py
"""组件提供者注册表 — 有比 DataSource 更通用的扩展模式。"""

class ProviderRegistry:
    """单例，但可被测试代码干净地覆盖。"""
    tokenizer: ITokenizer
    encoder: IEncoder
    router: IRouter
    value_net: IValueNetwork
    memory: IMemoryBank

    @classmethod
    def default(cls) -> "ProviderRegistry":
        return ProviderRegistry(
            tokenizer=QwenTokenizer(),
            encoder=TinyEncoder(),
            router=GumbelRouter(),
            value_net=ValueNetwork(),
            memory=MemoryBank(),
        )

    @classmethod
    def override(cls, **kwargs) -> "ProviderRegistry":
        """测试/离线用：只替换部分组件，其余保持默认。"""
        base = cls.default()
        for k, v in kwargs.items():
            setattr(base, k, v)
        return base

# ── 使用方式 (engine_provider.py) ──
registry = ProviderRegistry.default()

    model=model,

    memory_bank=registry.memory,
    encoder=registry.encoder,
    router=registry.router,
    value_net=registry.value_net,
)
